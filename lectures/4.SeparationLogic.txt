  private cv = 0;

Module CellA.
  def set(v) ≡
    v0 <- trigger (Take(nat));;
    trigger (Assume(cell(v0)));;
    trigger (Guarantee(cell(v)));;;
    Ret ().

  def get() ≡
    v0 <- trigger (Take(nat));;
    trigger (Assume(cell(v0)));;;
    r <- trigger (Choose(nat));;
    trigger (Guarantee(⌜r = v0⌝ * cell(v0)));;;
    Ret r.
    
End CellA.

forall v0, { cell(v0) } Cell.set(v) { cell(v)	 }
forall v0, { cell(v0) } r = Cell.get() { ⌜r = v0⌝ * cell(v0) }
